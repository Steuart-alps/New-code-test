Subject: [PATCH] Fix critical security/functionality issues from audit

1. Require auth on /api/storage endpoints (were fully unauthenticated) and
   enforce tenant-scoped ACL ownership on object downloads instead of the
   previously-commented-out check.
2. Block role self-escalation on PUT /api/users/:id - a client_admin could
   promote themselves (or another same-client user) to consultant, which was
   only blocked on user creation, not on edits.
3. Fix daily-track-am/pm entitlement gating: the router-level guard referenced
   service keys ("dailytrack_am"/"dailytrack_pm") that don't exist in the
   SERVICES catalog, permanently 403'ing these checklists for every client
   whose trial had ended. Replaced with requireAnyService(kitchentrack,
   safetrack) at the router level plus a per-request check against the
   specific service the checklistType belongs to.
4. Fix FixTrack photo/video upload: request-upload called
   getObjectEntityUploadURL() with arguments the real method doesn't accept,
   so uploadURL was always undefined and every upload failed. Now matches
   DocTrack's working pattern, plus tags each upload with its owning
   clientId so the download route's new ACL check doesn't break it.
5. Fix module-catalog drift between the marketing pricing builder and
   signup/trial-ended: landing.tsx used the key "premisestrack" for that
   module while the real backend key is "safetrack" (services.ts,
   ADDON_KEYS) - fixed. signup.tsx and trial-ended.tsx only recognized 3 of
   7 add-on modules and silently dropped the rest of a visitor's selection;
   expanded both to the full set with keys matching ADDON_KEYS.

+/**
+ * Route guard variant for routes that serve more than one billed branch (e.g.
+ * the daily AM/PM checklists cover both kitchen and premises items). Passes
+ * if the client is entitled to ANY of the given keys; individual handlers are
+ * still responsible for checking the specific key that matches the record
+ * being read/written (see requireAnyEntitlement below).
+ */
+export function requireAnyService(...keys: ServiceKey[]): RequestHandler {
+  return async (req, res, next) => {
+    try {
+      const clientId = getClientId(req);
+      if (!clientId) return res.status(400).json({ error: "No client context" });
+      const services = await getEntitledServices(clientId);
+      if (!keys.some((key) => isEntitled(services, key))) {
+        return res.status(403).json({
+          error: `None of the required services (${keys.map((k) => SERVICES[k].label).join(", ")}) are enabled for this account`,
+          code: "SERVICE_NOT_ENABLED",
+          service: keys[0],
+        });
+      }
+      next();
+    } catch (err) {
+      next(err);
+    }
+  };
+}
+
+/** Non-middleware check for use inside a handler once req.currentUser/clientId is known. */
+export async function requireAnyEntitlement(clientId: number, ...keys: ServiceKey[]): Promise<boolean> {
+  const services = await getEntitledServices(clientId);
+  return keys.some((key) => isEntitled(services, key));
+}
